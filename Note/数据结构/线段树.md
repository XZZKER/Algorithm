# 线段树

``````c++
template<typename T, typename Merge = std::plus<T>> class SegmentTree {
    std::vector<T> tree;
    Merge merge;

    void maintain(int node) {
        tree[node] = merge(tree[node * 2], tree[node * 2 + 1]);
    }

    void build(const std::vector<T>& init, int node, int l, int r) {
        if (l == r) {
            tree[node] = init[l];
            return;
        }
        int m = (l + r) >> 1;
        build(init, node * 2, l, m);
        build(init, node * 2 + 1, m + 1, r);
        maintain(node);
    }

public:
    SegmentTree(const std::vector<T> &init, const Merge &m = Merge()) : merge(m) {
        int n = init.size();
        tree.resize(n * 4);
        build(init, 1, 0, n - 1);
    }

    void update(int node, int l, int r, int i, const T &v) {
        if (l == r) {
            tree[node] = v;
            return;
        }
        int m = (l + r) >> 1;
        if (i <= m)
            update(node * 2, l, m, i, v);
        else
            update(node * 2 + 1, m + 1, r, i, v);
        maintain(node);
    }

    T query(int node, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr)
            return tree[node];
        int m = (l + r) >> 1;
        if (qr <= m)
            return query(node * 2, l, m, ql, qr);
        if (ql > m)
            return query(node * 2 + 1, m + 1, r, ql, qr);
        return merge(query(node * 2, l, m, ql, m), query(node * 2 + 1, m + 1, r, m + 1, qr));
    }
};
``````
