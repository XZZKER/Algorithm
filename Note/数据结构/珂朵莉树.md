# 珂朵莉树

## 模板

``````c++
template<typename T, typename F> class OldDriverTree {
    struct node {
        T l, r;
        mutable F v;
        node(const T l, const T r, const F v) : l(l), r(r), v(v) {}
        bool operator < (const node &other) const {
            return l < other.l;
        }
    };
    
    std::set<node> tree;
    
    auto cut(T pos) {
        auto it = tree.lower_bound(node(pos, 0, 0));
        if (it != tree.end() && it->l == pos)
            return it;
        it--;
        T l = it->l, r = it->r;
        F v = it->v;
        tree.erase(it);
        tree.insert(node(l, pos - 1, v));
        return tree.insert(node(pos, r, v)).first;
    }
    
public:
    OldDriverTree(const T l, const T r, const F v) {
        tree.insert(node(l, r, v));
    }
    
    void update(T l, T r, F v) {
        auto itr = cur(r + 1), itl = cur(l);
        tree.erase(itl, itr);
        tree.insert(node(l, r, v));
    }
};
``````
