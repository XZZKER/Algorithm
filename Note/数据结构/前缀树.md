# 前缀树

``````c++
class Trie {
    int idx, width;
    std::vector<std::vector<int>> tree;
    std::vector<int> ed;
    
public:
	Trie(int n, int w) : width(w), idx(1), (n, std::vector<int>(w)), ed(n) {}
    
    void insert(std::string s) {
        int cur = 1;
        for (int i = 0; i < s.size(); i++) {
            int path = s - 'a';
            if (tree[cur][path] == 0)
                tree[cur][path] = ++idx;
            cur = tree[cur][path];
        }
        ed[cur]++;
    }
    
    int search(std::string s) {
        int cur = 1;
        for (int i = 0; i < s.size(); i++) {
            int path = s - 'a';
            if (tree[cur][path] == 0)
                return 0;
            cur = tree[cur][path];
        }
        return ed[cur];
    }
};
``````
