# 拓扑排序

## 模板

``````c++
std::vector<int> TopoLogicalSort(int n, std::vector<std::vector<int>> edges) {
    std::vector<std::vector<int>> g(n);
    std::vector<int> pre(n);
    for (auto &edge : edges) {
        g[edge[0]].push_back(edge[1]);
        pre[edge[1]]++;
    }
    
    std::queue<int> q;
    for (int i = 0; i < n; i++) {
        if (pre[i] == n)
            q.push_back(i);
    }
    
    std::vector<int> topo;
    while (q.size()) {
        int u = q.front();
        q.pop();
        topo.push_back(u);
        for (int v : g[u]) {
            if (--pre[v] == 0)
                q.push(v);
        }
    }
    
    return topo;
}
``````
